import customtkinter as ctk # This creates a 'Frame' (like a
in HTML) class AccountCard(ctk.CTkFrame): def __init__(self, master): super().__init__(master) # This is like a Label in HTML self.label = ctk.CTkLabel(self, text="Gmail Account", font=("Arial", 20)) self.label.pack(pady=10) # This is like an Input field in HTML self.entry = ctk.CTkEntry(self, placeholder_text="Enter password") self.entry.pack(pady=5, padx=20) # This is a Button self.btn = ctk.CTkButton(self, text="Save Password") self.btn.pack(pady=10)
